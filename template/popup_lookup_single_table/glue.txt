function showForm${{parent_controller}}(o, n) {
  var f = document.body._form, g = f.grid;
  if (f._action == 'View') return;
  // AGENT: chặn điều kiện nghiệp vụ nếu cần (vd đã chọn chứng từ con thì không đổi)
  if (g && g._memvars) g._originMemvars = g._memvars;
  g.showForm('{{identity}}Filter');
}
function clear${{parent_controller}}(o, n) {
  var f = document.body._form;
  if (!f || f._action == 'View') return;
  f.setItemValues('{{dest_id_field}}, {{dest_no_field}}', ['', '']);
  // AGENT: xóa thêm các field/grid đã load theo chứng từ nguồn
  f.focus('{{dest_no_field}}');
}
